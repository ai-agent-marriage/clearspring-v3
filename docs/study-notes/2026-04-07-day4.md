```javascript
// pages/order/create/create.js
Page({
  data: {
    basePrice: 10, // 每份基础价格（元）
    quantity: 1,
    serviceItems: [],
    totalAmount: 0
  },
  
  onQuantityChange(e) {
    this.setData({ quantity: e.detail });
    this.calculateTotal();
  },

    this.setData({ serviceItems: e.detail });
    this.calculateTotal();

  calculateTotal() {
    const { basePrice, quantity, serviceItems } = this.data;
    
    // 基础金额
    const baseAmount = basePrice * quantity;
    
    // 增值服务金额
    const serviceAmount = serviceItems.reduce((sum, item) => {
      return sum + item.price;
    }, 0);
    
    // 总金额
    const totalAmount = baseAmount + serviceAmount;
    
    this.setData({ 
      totalAmount,
      baseAmount,
      serviceAmount
    });
  },
  
  onSubmit() {
    const { totalAmount, selectedDate, isSuitForProtect } = this.data;
    
    // 验证
    if (!selectedDate) {
      wx.showToast({ title: '请选择护生日期', icon: 'none' });

        success: (res) => {
          if (res.confirm) {

      return;
```

#### 支付流程实现

```javascript
// pages/order/confirm/confirm.js
Page({
  data: {
    orderNo: '',
    orderInfo: null,
    agreedTerms: false,
    agreedPrivacy: false
  },
  
  onLoad(options) {
    this.setData({ orderNo: options.orderNo });
    this.loadOrderInfo();
  },
  
  async loadOrderInfo() {
    const res = await wx.request({
      url: `/api/order/detail/${this.data.orderNo}`,
      method: 'GET'
    });
    this.setData({ orderInfo: res.data.data });
  },
  
  onAgreementsChange(e) {
    const values = e.detail.value;
    this.setData({
      agreedTerms: values.includes('terms'),
      agreedPrivacy: values.includes('privacy')
    });
  },
  
  async onPay() {
    const { agreedTerms, agreedPrivacy, orderInfo } = this.data;
    
    // 验证协议勾选
    if (!agreedTerms || !agreedPrivacy) {
      wx.showToast({
        title: '请先同意相关协议',
        icon: 'none'
      });
      return;
    }
    
    wx.showLoading({ title: '支付中...' });
    
    try {
      // 1. 调用后端获取支付参数
      const payRes = await wx.request({
        url: '/api/order/pay',
        method: 'POST',
        data: { orderNo: this.data.orderNo }
      });
      
      const payParams = payRes.data.data;
      
      // 2. 调起微信支付
      await wx.requestPayment({
        timeStamp: payParams.timeStamp,
        nonceStr: payParams.nonceStr,
        package: payParams.package,
        signType: payParams.signType,
        paySign: payParams.paySign
      });
      
      wx.hideLoading();
      
      // 3. 支付成功，跳转到订单详情页
      wx.showModal({
        title: '支付成功',
        content: '订单已支付成功',
        showCancel: false,
        success: () => {
          wx.redirectTo({
            url: `/pages/order/detail/detail?orderNo=${this.data.orderNo}`
          });
        }
      });
    } catch (error) {
      wx.hideLoading();
      
      if (error.errMsg.includes('cancel')) {
        wx.showToast({ title: '支付已取消', icon: 'none' });
      } else {
        wx.showToast({ title: '支付失败', icon: 'none' });
      }
    }
  }
});
```

#### 状态流转验证

```java
// service/OrderService.java
@Service
public class OrderService {
    
    // 定义合法状态流转
    private static final Map<OrderStatus, Set<OrderStatus>> VALID_TRANSITIONS = Map.ofEntries(
        Map.entry(OrderStatus.CREATED, Set.of(OrderStatus.PENDING_PAYMENT, OrderStatus.CANCELLED)),
        Map.entry(OrderStatus.PENDING_PAYMENT, Set.of(OrderStatus.PAID, OrderStatus.CANCELLED)),
        Map.entry(OrderStatus.PAID, Set.of(OrderStatus.ACCEPTED)),
        Map.entry(OrderStatus.ACCEPTED, Set.of(OrderStatus.EXECUTING)),
        Map.entry(OrderStatus.EXECUTING, Set.of(OrderStatus.COMPLETED)),
        Map.entry(OrderStatus.COMPLETED, Set.of(OrderStatus.REVIEWING))
    );
    
    /**
     * 验证状态流转是否合法
     */
    private boolean isValidTransition(OrderStatus from, OrderStatus to) {
        Set<OrderStatus> validTargets = VALID_TRANSITIONS.get(from);
        return validTargets != null && validTargets.contains(to);
    }
    
    /**
     * 更新订单状态
     */
    @Transactional(rollbackFor = Exception.class)
    public Order updateOrderStatus(String orderNo, OrderStatus newStatus, String remark) {
        Order order = orderMapper.selectByOrderNo(orderNo);
        if (order == null) {
            throw new BusinessException("订单不存在");
        }
        
        OrderStatus oldStatus = order.getStatus();
        
        // 验证状态流转合法性
        if (!isValidTransition(oldStatus, newStatus)) {
            throw new BusinessException(
                String.format("订单状态流转非法：%s -> %s", oldStatus, newStatus)
            );
        }
        
        // 更新订单状态
        order.setStatus(newStatus);
        order.setUpdateTime(LocalDateTime.now());
        orderMapper.updateById(order);
        
        // 记录状态流转日志
        OrderStatusLog log = new OrderStatusLog();
        log.setOrderNo(orderNo);
        log.setOldStatus(oldStatus);
        log.setNewStatus(newStatus);
        log.setRemark(remark);
        log.setCreateTime(LocalDateTime.now());
        orderStatusLogMapper.insert(log);
        
        return order;
    }
}
```

```java
// controller/OrderController.java
@RestController
@RequestMapping("/api/order")
public class OrderController {
    
    @Autowired
    private OrderService orderService;

        @PathVariable String orderNo,
        @RequestBody(required = false) CancelOrderDTO dto

// service/OrderService.java
@Service
public class OrderService {
    
    @Autowired
    private OrderMapper orderMapper;
    
    @Autowired
    private PaymentService paymentService;
    
    /**
     * 取消订单
     */
    @Transactional(rollbackFor = Exception.class)
    public OrderResult cancelOrder(String orderNo, CancelOrderDTO dto) {
        Order order = orderMapper.selectByOrderNo(orderNo);
        if (order == null) {
            throw new BusinessException("订单不存在");
        }
        
        // 验证订单状态（仅待支付状态可取消）
        if (order.getStatus() != OrderStatus.PENDING_PAYMENT) {
            throw new BusinessException("仅待支付状态的订单可取消");
        }
        
        // 更新订单状态
        order.setStatus(OrderStatus.CANCELLED);
        order.setCancelReason(dto != null ? dto.getReason() : null);
        order.setCancelTime(LocalDateTime.now());
        order.setUpdateTime(LocalDateTime.now());
        orderMapper.updateById(order);
        
        // 记录状态流转日志
        logStatusChange(orderNo, OrderStatus.PENDING_PAYMENT, OrderStatus.CANCELLED, "用户取消");
        
        // 如果已支付，触发退款流程
        if (order.getPaymentStatus() == PaymentStatus.PAID) {
            paymentService.refund(orderNo);
        }
        
        // 发送取消通知
        notificationService.sendOrderCancelled(orderNo);
        
        return convertToResult(order);
    }
}
```

```java
@Slf4j
public class OrderCancelTask {
    
    @Autowired
    private OrderService orderService;
    
    @Autowired
    private OrderMapper orderMapper;
    
    /**
     * 每 30 分钟执行一次
     * 扫描待支付超过 48 小时的订单，自动取消
     */
    @Scheduled(cron = "0 */30 * * * ?")
    public void autoCancelExpiredOrders() {
        log.info("开始执行订单自动取消任务");
        
        try {
            // 计算 48 小时前的时间
            LocalDateTime expireTime = LocalDateTime.now().minusHours(48);
            
            // 查询待支付超过 48 小时的订单
            List<Order> expiredOrders = orderMapper.selectExpiredOrders(
                OrderStatus.PENDING_PAYMENT, 
                expireTime
            );
            
            log.info("找到 {} 个过期订单", expiredOrders.size());
            
            // 批量取消
            for (Order order : expiredOrders) {
                try {
                    CancelOrderDTO dto = new CancelOrderDTO();
                    dto.setReason("48 小时未支付，系统自动取消");
                    orderService.cancelOrder(order.getOrderNo(), dto);

                } catch (Exception e) {
                    log.error("自动取消订单失败：{}", order.getOrderNo(), e);
                }
            }
            
            log.info("订单自动取消任务执行完成，成功{}个", expiredOrders.size());
        } catch (Exception e) {
            log.error("订单自动取消任务执行异常", e);
```

```java
        notificationService.sendOrderAccepted(orderNo, org.getOrgName());
```

```java
@Service
@Slf4j
public class SettlementService {
    
    @Autowired
    private SettlementMapper settlementMapper;
    
    @Autowired
    private OrderService orderService;
    
    @Autowired
    private PaymentService paymentService;
    
    /**
     * 创建结算单
     */
    @Transactional(rollbackFor = Exception.class)
    public SettlementResult createSettlement(SettlementCreateDTO dto) {
        // 1. 验证订单状态（必须为已完成）
        Order order = orderService.validateOrderStatus(dto.getOrderNo(), OrderStatus.COMPLETED);
        
        // 2. 验证机构
        Org org = orgMapper.selectById(dto.getOrgId());
        if (org == null) {
            throw new BusinessException("机构不存在");
        }
        
        // 3. 检查是否已创建结算单
        Settlement existing = settlementMapper.selectByOrderNo(dto.getOrderNo());
        if (existing != null) {
            throw new BusinessException("结算单已存在");
        }
        
        // 4. 计算结算金额
        BigDecimal totalAmount = order.getTotalAmount();
        BigDecimal platformFeeRate = new BigDecimal("0.1"); // 10% 平台费
        BigDecimal platformFee = totalAmount.multiply(platformFeeRate).setScale(2, RoundingMode.HALF_UP);
        BigDecimal settlementAmount = totalAmount.subtract(platformFee);
        
        // 5. 创建结算单
        Settlement settlement = new Settlement();
        settlement.setSettlementNo(generateSettlementNo());
        settlement.setOrderNo(dto.getOrderNo());
        settlement.setOrgId(dto.getOrgId());
        settlement.setOrgName(org.getOrgName());
        settlement.setSettlementAmount(settlementAmount);
        settlement.setServiceFee(dto.getServiceFee());
        settlement.setPlatformFee(platformFee);
        settlement.setTotalAmount(totalAmount);
        settlement.setStatus(SettlementStatus.PENDING_CONFIRM);
        settlement.setRemark(dto.getRemark());
        settlementMapper.insert(settlement);
        
        log.info("结算单创建成功：settlementNo={}, orderNo={}", settlement.getSettlementNo(), dto.getOrderNo());
        
        return convertToResult(settlement);
    }
    
    /**
     * 确认结算
     */
    @Transactional(rollbackFor = Exception.class)
    public SettlementResult confirmSettlement(Long id, Long confirmBy) {
        Settlement settlement = settlementMapper.selectById(id);
        if (settlement == null) {
            throw new BusinessException("结算单不存在");
        }
        
        // 验证状态
        if (settlement.getStatus() != SettlementStatus.PENDING_CONFIRM) {
            throw new BusinessException("仅待确认状态的结算单可确认");
        }
        
        // 更新状态
        settlement.setStatus(SettlementStatus.CONFIRMED);
        settlement.setConfirmBy(confirmBy);
        settlement.setConfirmTime(LocalDateTime.now());
        settlementMapper.updateById(settlement);
        
        // 触发打款流程
        paymentService.schedulePayment(settlement);
        
        log.info("结算确认成功：settlementNo={}", settlement.getSettlementNo());
        
        return convertToResult(settlement);
    }
    
    private String generateSettlementNo() {
        return "SETTLE" + LocalDateTime.now().format(DateTimeFormatter.ofPattern("yyyyMMdd")) 
               + RandomUtil.randomNumbers(6);
    }
```

## 💡 遇到的问题与解决方案

### 问题 1：订单状态并发修改

**现象**：多个请求同时修改订单状态，导致状态不一致

**原因**：未加锁，乐观锁未生效

**解决**：
1. 使用数据库乐观锁（version 字段）
2. 关键状态流转加分布式锁（Redis）
3. 状态变更日志记录（便于追溯）

```java
// 乐观锁实现
@Version
private Integer version;

// 分布式锁实现
public OrderResult cancelOrder(String orderNo, CancelOrderDTO dto) {
    String lockKey = "order:lock:" + orderNo;
    RLock lock = redissonClient.getLock(lockKey);
    
    if (lock.tryLock(3, 10, TimeUnit.SECONDS)) {
        try {
            // 执行取消逻辑
            return doCancelOrder(orderNo, dto);
        } finally {
            lock.unlock();
        }
    } else {
        throw new BusinessException("操作过于频繁，请稍后重试");
    }
}
```

### 问题 2：实时计算精度问题

**现象**：金额计算出现 0.01 元误差

**原因**：使用 double 类型计算金额

**解决**：
1. 所有金额使用 BigDecimal 类型
2. 设置精度为 2，四舍五入
3. 前端显示使用分（整数），展示时转换为元

```java
// 正确做法
BigDecimal basePrice = new BigDecimal("10.00");
BigDecimal quantity = new BigDecimal("50");
BigDecimal baseAmount = basePrice.multiply(quantity).setScale(2, RoundingMode.HALF_UP);

// 错误做法（会导致精度丢失）
double basePrice = 10.00;
double quantity = 50;
double baseAmount = basePrice * quantity; // 可能出现精度问题
```

### 问题 3：机构承接订单超卖

**现象**：多个机构同时承接同一订单

**原因**：承接操作未加锁

**解决**：
1. 承接前检查订单状态（必须为已支付）
2. 使用数据库唯一索引（order_no + org_id）
3. 承接操作加行锁（SELECT ... FOR UPDATE）

```java
@Transactional
public OrgOrderResult acceptOrder(String orderNo, OrgAcceptDTO dto) {
    // 加行锁查询订单
    Order order = orderMapper.selectByOrderNoForUpdate(orderNo);
    
    // 双重检查状态
    if (order.getStatus() != OrderStatus.PAID) {
        throw new BusinessException("订单状态异常");
    }
    
    // 检查是否已被承接
    OrgOrder existing = orgOrderMapper.selectByOrderNo(orderNo);
    if (existing != null) {
        throw new BusinessException("订单已被承接");
    }
    
    // 继续承接逻辑...
}
```
